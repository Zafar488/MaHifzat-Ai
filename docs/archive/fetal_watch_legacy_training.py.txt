Three-class classifier (Normal / Suspect / Pathological) trained on
the UCI Cardiotocography dataset with LightGBM (primary), Random Forest
(baseline), and XGBoost (comparison).
"""

# ═══════════════════════════════════════════════════════════════════════════
# Helper utilities
# ═══════════════════════════════════════════════════════════════════════════

def _macro_f1(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    """Compute macro-averaged F1 from sklearn's classification_report."""
    report = classification_report(y_true, y_pred, output_dict=True, zero_division=0)
    return float(report["macro avg"]["f1-score"])


def _evaluate_model(model, X_test: pd.DataFrame, y_test: pd.Series, label: str) -> dict:
    """Run full evaluation on the held-out test set and return a metrics dict."""
    y_pred = model.predict(X_test)

    report = classification_report(y_test, y_pred, output_dict=True, zero_division=0)
    macro_f1 = float(report["macro avg"]["f1-score"])

    # AUROC — one-vs-rest (needs probability estimates)
    try:
        y_prob = model.predict_proba(X_test)
        auroc = float(
            roc_auc_score(
                pd.get_dummies(y_test).values,
                y_prob,
                multi_class="ovr",
                average="macro",
            )
        )
    except Exception:
        auroc = None

    # Per-class metrics
    classes = sorted(y_test.unique())
    per_class: dict = {}
    for cls in classes:
        cls_key = str(cls)
        per_class[cls_key] = {
            "precision": float(report.get(cls_key, {}).get("precision", 0.0)),
            "recall": float(report.get(cls_key, {}).get("recall", 0.0)),
            "f1-score": float(report.get(cls_key, {}).get("f1-score", 0.0)),
        }

    print(f"\n  [{label}] Evaluation on test set:")
    print(f"    Macro-F1 : {macro_f1:.4f}")
    if auroc is not None:
        print(f"    AUROC    : {auroc:.4f}")
    print(classification_report(y_test, y_pred, zero_division=0))

    return {
        "macro_f1": macro_f1,
        "auroc_ovr_macro": auroc,
        "per_class": per_class,
        "confusion_matrix": confusion_matrix(y_test, y_pred).tolist(),
        "classification_report": report,
    }


# ═══════════════════════════════════════════════════════════════════════════
# LightGBM — Optuna HPO
# ═══════════════════════════════════════════════════════════════════════════

def _train_lightgbm(
    X_train: pd.DataFrame,
    y_train: pd.Series,
    X_val: pd.DataFrame,
    y_val: pd.Series,
    n_trials: int = 50,
) -> tuple:
    """Train LightGBM with Optuna hyper-parameter search."""

    # Encode labels for CV scoring consistency
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)

    def objective(trial: optuna.Trial) -> float:
        params = {
            "num_leaves": trial.suggest_int("num_leaves", 20, 150),
            "learning_rate": trial.suggest_float("learning_rate", 0.01, 0.3, log=True),
            "n_estimators": trial.suggest_int("n_estimators", 100, 1000),
            "min_child_samples": trial.suggest_int("min_child_samples", 5, 50),
            "subsample": trial.suggest_float("subsample", 0.6, 1.0),
            "colsample_bytree": trial.suggest_float("colsample_bytree", 0.6, 1.0),
            "reg_alpha": trial.suggest_float("reg_alpha", 0.0, 10.0),
            "reg_lambda": trial.suggest_float("reg_lambda", 0.0, 10.0),
            "class_weight": "balanced",
            "random_state": 42,
            "verbose": -1,
            "n_jobs": -1,
        }
        clf = lgb.LGBMClassifier(**params)
        scores = cross_val_score(
            clf, X_train, y_train,
            cv=skf,
            scoring="f1_macro",
            n_jobs=-1,
        )
        return float(scores.mean())

    print("[LightGBM] Starting Optuna HPO ({} trials) ...".format(n_trials))
    study = optuna.create_study(
        direction="maximize",
        pruner=optuna.pruners.MedianPruner(n_startup_trials=5, n_warmup_steps=5),
        study_name="fetalwatch_lgbm",
    )
    study.optimize(objective, n_trials=n_trials, show_progress_bar=False)

    best_params = study.best_params
    best_params.update({
        "class_weight": "balanced",
        "random_state": 42,
        "verbose": -1,
        "n_jobs": -1,
    })
    print(f"[LightGBM] Best CV macro-F1: {study.best_value:.4f}")
    print(f"[LightGBM] Best params: {best_params}")

    # Retrain on full training set (+ early stopping on validation)
    best_model = lgb.LGBMClassifier(**best_params)
    best_model.fit(
        X_train, y_train,
        eval_set=[(X_val, y_val)],
        callbacks=[
            lgb.early_stopping(stopping_rounds=30, verbose=False),
            lgb.log_evaluation(period=0),
        ],
    )
    return best_model, best_params


# ═══════════════════════════════════════════════════════════════════════════
# Random Forest — baseline
# ═══════════════════════════════════════════════════════════════════════════

def _train_random_forest(
    X_train: pd.DataFrame,
    y_train: pd.Series,
) -> tuple:
    """Train a Random Forest baseline (no HPO)."""
    print("[RandomForest] Training baseline (n_estimators=500, balanced) ...")
    rf = RandomForestClassifier(
        n_estimators=500,
        class_weight="balanced",
        random_state=42,
        n_jobs=-1,
    )
    rf.fit(X_train, y_train)

    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    cv_scores = cross_val_score(rf, X_train, y_train, cv=skf, scoring="f1_macro", n_jobs=-1)
    print(f"[RandomForest] CV macro-F1: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
    params = {"n_estimators": 500, "class_weight": "balanced", "random_state": 42}
    return rf, params


# ═══════════════════════════════════════════════════════════════════════════
# XGBoost — Optuna HPO
# ═══════════════════════════════════════════════════════════════════════════

def _train_xgboost(
    X_train: pd.DataFrame,
    y_train: pd.Series,
    X_val: pd.DataFrame,
    y_val: pd.Series,
    n_trials: int = 30,
) -> tuple:
    """Train XGBoost with Optuna hyper-parameter search."""

    # XGBoost multi_class requires integer labels [0..n-1]
    classes = sorted(y_train.unique())
    label_map = {cls: idx for idx, cls in enumerate(classes)}
    inv_label_map = {idx: cls for cls, idx in label_map.items()}

    y_train_enc = y_train.map(label_map).astype(int)
    y_val_enc = y_val.map(label_map).astype(int)

    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)

    def objective(trial: optuna.Trial) -> float:
        params = {
            "max_depth": trial.suggest_int("max_depth", 3, 10),
            "learning_rate": trial.suggest_float("learning_rate", 0.01, 0.3, log=True),
            "n_estimators": trial.suggest_int("n_estimators", 100, 800),
            "min_child_weight": trial.suggest_int("min_child_weight", 1, 10),
            "subsample": trial.suggest_float("subsample", 0.6, 1.0),
            "colsample_bytree": trial.suggest_float("colsample_bytree", 0.6, 1.0),
            "objective": "multi:softprob",
            "num_class": len(classes),
            "eval_metric": "mlogloss",
            "random_state": 42,
            "verbosity": 0,
            "n_jobs": -1,
        }
        clf = xgb.XGBClassifier(**params)
        scores = cross_val_score(
            clf, X_train, y_train_enc,
            cv=skf,
            scoring="f1_macro",
            n_jobs=-1,
        )
        return float(scores.mean())

    print("[XGBoost] Starting Optuna HPO ({} trials) ...".format(n_trials))
    study = optuna.create_study(
        direction="maximize",
        pruner=optuna.pruners.MedianPruner(n_startup_trials=5, n_warmup_steps=5),
        study_name="fetalwatch_xgb",
    )
    study.optimize(objective, n_trials=n_trials, show_progress_bar=False)

    best_params = study.best_params
    best_params.update({
        "objective": "multi:softprob",
        "num_class": len(classes),
        "eval_metric": "mlogloss",
        "random_state": 42,
        "verbosity": 0,
        "n_jobs": -1,
    })
    print(f"[XGBoost] Best CV macro-F1: {study.best_value:.4f}")
    print(f"[XGBoost] Best params: {best_params}")

    # Retrain on full training set
    best_model = xgb.XGBClassifier(**best_params)
    best_model.fit(
        X_train, y_train_enc,
        eval_set=[(X_val, y_val_enc)],
        verbose=False,
    )
    return best_model, best_params, inv_label_map


# ═══════════════════════════════════════════════════════════════════════════
# Main training pipeline
# ═══════════════════════════════════════════════════════════════════════════

def train_fetal_watch(
    data_dir: str | None = None,
    n_trials: int = 50,
) -> dict:
    """Run the full FetalWatch training pipeline.

    Parameters
    ----------
    data_dir : str, optional
        Directory that contains ``uci_ctg.parquet``.
    n_trials : int
        Number of Optuna trials for LightGBM HPO (XGBoost uses 60 % of this).

    Returns
    -------
    results : dict
        Nested dictionary with evaluation metrics for every model.
    """
    if data_dir is None:
        data_dir = os.path.join(_PROJECT_ROOT, "data")

    # ------------------------------------------------------------------
    # 1. Load data
    # ------------------------------------------------------------------
    print("\n" + "=" * 60)
    print("  FetalWatch — Fetal Health Classification Pipeline")
    print("=" * 60)
    print("\n[Step 1] Loading UCI CTG dataset ...")
    X, y = load_uci_ctg(data_dir=data_dir)
    print(f"  Loaded {len(X)} samples, {X.shape[1]} features")
    print(f"  Class distribution:\n{y.value_counts().to_string()}")

    # ------------------------------------------------------------------
    # 2. Preprocess: split + scale + SMOTE
    # ------------------------------------------------------------------
    print("\n[Step 2] Preprocessing (split 70/15/15 + RobustScaler + SMOTE) ...")
    prep = preprocess_pipeline(
        X, y,
        scale=True,
        balance="smote",
        test_size=0.15,
        val_size=0.15,
        random_state=42,
        scale_method="robust",
    )
    X_train, X_val, X_test = prep["X_train"], prep["X_val"], prep["X_test"]
    y_train, y_val, y_test = prep["y_train"], prep["y_val"], prep["y_test"]
    print(f"  Train: {len(X_train)} | Val: {len(X_val)} | Test: {len(X_test)}")
    print(f"  Train class distribution after SMOTE:\n{y_train.value_counts().to_string()}")

    results: dict = {}

    # ------------------------------------------------------------------
    # 3a. LightGBM (primary)
    # ------------------------------------------------------------------
    print("\n[Step 3a] Training LightGBM (Optuna HPO) ...")
    lgbm_model, lgbm_params = _train_lightgbm(X_train, y_train, X_val, y_val, n_trials=n_trials)
    lgbm_metrics = _evaluate_model(lgbm_model, X_test, y_test, label="LightGBM")
    results["LightGBM"] = {"params": lgbm_params, **lgbm_metrics}

    # ------------------------------------------------------------------
    # 3b. Random Forest (baseline)
    # ------------------------------------------------------------------
    print("\n[Step 3b] Training Random Forest (baseline) ...")
    rf_model, rf_params = _train_random_forest(X_train, y_train)
    rf_metrics = _evaluate_model(rf_model, X_test, y_test, label="RandomForest")
    results["RandomForest"] = {"params": rf_params, **rf_metrics}

    # ------------------------------------------------------------------
    # 3c. XGBoost (comparison)
    # ------------------------------------------------------------------
    print("\n[Step 3c] Training XGBoost (Optuna HPO) ...")
    xgb_trials = max(10, int(n_trials * 0.6))
    xgb_model, xgb_params, xgb_inv_label_map = _train_xgboost(
        X_train, y_train, X_val, y_val, n_trials=xgb_trials
    )

    # XGBoost predictions return integer labels — map back to strings
    class _XGBWrapper:
        """Thin wrapper so _evaluate_model sees string labels."""

        def __init__(self, model, inv_map):
            self._model = model
            self._inv_map = inv_map

        def predict(self, X):
            raw = self._model.predict(X)
            return np.array([self._inv_map[int(r)] for r in raw])

        def predict_proba(self, X):
            return self._model.predict_proba(X)

    xgb_wrapped = _XGBWrapper(xgb_model, xgb_inv_label_map)
    xgb_metrics = _evaluate_model(xgb_wrapped, X_test, y_test, label="XGBoost")
    results["XGBoost"] = {"params": xgb_params, **xgb_metrics}

    # ------------------------------------------------------------------
    # 4. Comparison table
    # ------------------------------------------------------------------
    print("\n" + "=" * 60)
    print("  MODEL COMPARISON")
    print("=" * 60)
    header = f"{'Model':<15} {'Macro-F1':>10} {'AUROC':>10}"
    print(header)
    print("-" * len(header))
    for name, m in results.items():
        auroc_str = f"{m['auroc_ovr_macro']:.4f}" if m["auroc_ovr_macro"] is not None else "N/A"
        print(f"{name:<15} {m['macro_f1']:>10.4f} {auroc_str:>10}")

    # ------------------------------------------------------------------
    # 5. Select & save best model
    # ------------------------------------------------------------------
    best_name = max(results, key=lambda k: results[k]["macro_f1"])
    best_model_obj = {
        "LightGBM": lgbm_model,
        "RandomForest": rf_model,
        "XGBoost": xgb_model,
    }[best_name]

    out_dir = os.path.join(_PROJECT_ROOT, "artifacts", "fetalwatch")
    os.makedirs(out_dir, exist_ok=True)

    model_path = os.path.join(out_dir, "best_model.joblib")
    joblib.dump(best_model_obj, model_path)
    print(f"\n[Saved] Best model '{best_name}' -> {model_path}")

    results_path = os.path.join(out_dir, "results.json")
    # Make results JSON-serialisable (drop numpy types)
    def _sanitize(obj):
        if isinstance(obj, (np.integer,)):
            return int(obj)
        if isinstance(obj, (np.floating,)):
            return float(obj)
        if isinstance(obj, np.ndarray):
            return obj.tolist()
        if isinstance(obj, dict):
            return {k: _sanitize(v) for k, v in obj.items()}
        if isinstance(obj, (list, tuple)):
            return [_sanitize(v) for v in obj]
        return obj

    serializable_results = _sanitize(results)
    serializable_results["best_model"] = best_name
    with open(results_path, "w", encoding="utf-8") as fp:
        json.dump(serializable_results, fp, indent=2)
    print(f"[Saved] Results -> {results_path}")

    print("\n" + "=" * 60)
    print(f"  BEST MODEL : {best_name}")
    print(f"  Macro-F1   : {results[best_name]['macro_f1']:.4f}")
    print("=" * 60 + "\n")

    return results


# ═══════════════════════════════════════════════════════════════════════════
# Entry point
# ═══════════════════════════════════════════════════════════════════════════

def main() -> None:
    """Run the full FetalWatch pipeline and print a formatted summary."""
    results = train_fetal_watch()

    # Formatted per-class sensitivity summary
    print("\n" + "=" * 60)
    print("  PER-CLASS SENSITIVITY (Recall) — Test Set")
    print("=" * 60)

    best_name = max(results, key=lambda k: results[k]["macro_f1"])
    print(f"\n  Best model: {best_name}")
    print(f"  Macro-F1  : {results[best_name]['macro_f1']:.4f}")
    auroc = results[best_name].get("auroc_ovr_macro")
    if auroc is not None:
        print(f"  AUROC     : {auroc:.4f}")

    print("\n  Per-class sensitivity (all models):")
    for model_name, metrics in results.items():
        print(f"\n  [{model_name}]")
        for cls, cls_metrics in metrics.get("per_class", {}).items():
            print(
                f"    {cls:<15}  "
                f"Precision={cls_metrics['precision']:.3f}  "
                f"Recall={cls_metrics['recall']:.3f}  "
                f"F1={cls_metrics['f1-score']:.3f}"
            )


if __name__ == "__main__":
    main()
